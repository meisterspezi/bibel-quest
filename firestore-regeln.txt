rules_version = '2';
// Sicherheitsregeln für die Gemeinde-Rangliste der Bibelbuch-Quest (Version 7.3 – Sammlung bethlehem (Weg nach Bethlehem, Advent); 6.95 – Sammlung talente (Talente-Zählung, nur der Verwalter liest); 6.88 – xpquellen: Regel vereinfacht (Limit 1000 Auswertungen pro Schreibzugriff, ab der 4. Quelle wurde abgelehnt); 6.80 – Sammlungen bibversuch (Biblionär-Wochenrunde einmal je Serverwoche) und spendenxp (XP für Spenden einmal je Talent); 6.79 – xpquellen: neue Quelle spende (Stiftshütte); 6.78 – Sammlung tagestruhe (Schatztruhe einmal je Serverdatum); 6.76 – xpquellen: neue Quelle abgleich; 6.75 – Sammlung xpprotokoll (Einzelprotokoll der XP-Vergaben, nur anlegen); 6.74 – xpquellen: Protokoll der täglichen Belohnung „bel“; 6.73 – Sammlung xpquellen (XP je Quelle); 6.68 – Sammlung stiftshuette; 6.65 – Geräteliste bis 20; 6.61 – Verwalter nur über Geräteliste einstellungen/entwickler.geraete; 6.59 – Beitreten ohne erneute Werteprüfung, Verwalter-Profil nicht per Auswahl; 6.40 – Wer wird Biblionär?: Duell-Art quiz, Rekord bib, Sammlung biblionaer; seit 6.37 bleiben fertige Duelle erhalten)
service cloud.firestore {
  match /databases/{database}/documents {

    function angemeldet() { return request.auth != null; }
    function ich() { return request.auth.uid; }
    function spielerPfad(pid) { return /databases/$(database)/documents/spieler/$(pid); }

    // Gehört das Profil pid zu mir? (neue Profile: Liste "owners", alte Profile: Dokument-ID = meine Kennung)
    function besitzer(pid) {
      return exists(spielerPfad(pid)) && (
        ('owners' in get(spielerPfad(pid)).data && ich() in get(spielerPfad(pid)).data.owners)
        || (!('owners' in get(spielerPfad(pid)).data) && ich() == pid));
    }

    function zahl(d, f, max) { return !(f in d) || (d[f] is int && d[f] >= 0 && d[f] <= max); }
    function text(d, f, max) { return !(f in d) || (d[f] is string && d[f].size() <= max); }

    // Gültiger Ranglisten-Eintrag (alte App-Versionen senden nur einen Teil der Felder – das ist erlaubt)
    function felder() {
      return ['name','xp','level','bestChain','chainAT','chainNT','chainRueck','stars','badges','badgeIds',
              'streak','bestStreak','blitz','combo','week','weekXp','avatar','frame','title','armor','atP','ntP','tl','t28','sg','bew','von','owners','code','updated',
              'duS','duG','bib'];
    }
    function gueltig(d) {
      return d.keys().hasOnly(felder()) && werteOk(d);
    }
    // Der Verwalter darf zusätzlich „zu“ setzen (= Eintrag wurde mit einem anderen zusammengeführt)
    function zuOk(d) {
      return !('zu' in d) || (d.zu is string && d.zu.size() <= 64);
    }
    function werteOk(d) {
      return d.name is string && d.name.size() >= 2 && d.name.size() <= 20
        && d.xp is int && d.xp >= 0 && d.xp <= 10000000
        && d.level is int && d.level >= 1 && d.level <= 1000
        && d.week is string && d.week.size() <= 10
        && zahl(d, 'weekXp', 1000000) && zahl(d, 'bestChain', 65) && zahl(d, 'chainAT', 38) && zahl(d, 'chainNT', 26)
        && zahl(d, 'chainRueck', 65) && zahl(d, 'stars', 24) && zahl(d, 'badges', 100) && zahl(d, 'streak', 100000)
        && zahl(d, 'bestStreak', 100000) && zahl(d, 'blitz', 1000) && zahl(d, 'combo', 1000) && zahl(d, 'armor', 6) && zahl(d, 'atP', 100) && zahl(d, 'ntP', 100) && zahl(d, 't28', 31) && zahl(d, 'bew', 16) && text(d, 'von', 64) && zahl(d, 'duS', 100000) && zahl(d, 'duG', 100000)
        && zahl(d, 'bib', 1000000)
        && text(d, 'avatar', 20) && text(d, 'frame', 20) && text(d, 'title', 30) && text(d, 'code', 12)
        && (!('badgeIds' in d) || (d.badgeIds is list && d.badgeIds.size() <= 80))
        && (!('tl' in d) || (d.tl is list && d.tl.size() <= 31))
        && (!('sg' in d) || (d.sg is list && d.sg.size() <= 8))
        && (!('owners' in d) || (d.owners is list && d.owners.size() <= 20));
    }

    function bisherigeBesitzer() { return 'owners' in resource.data ? resource.data.owners : []; }
    // Niemand darf aus der Besitzerliste entfernt werden (schützt auch vor alten App-Versionen)
    function besitzerBleiben() {
      return !('owners' in resource.data)
        || ('owners' in request.resource.data && request.resource.data.owners.hasAll(resource.data.owners));
    }
    // Hinzukommen darf höchstens das eigene Gerät (und die ursprüngliche Profil-Kennung)
    function hoechstensIchDazu(pid) {
      return !('owners' in request.resource.data)
        || request.resource.data.owners.hasOnly(bisherigeBesitzer().concat([ich(), pid]));
    }
    function istBesitzer(pid) {
      return ('owners' in resource.data && ich() in resource.data.owners)
        || (!('owners' in resource.data) && ich() == pid);
    }
    // „Ich spiele schon mit“: jedes angemeldete Gerät darf sich selbst als Besitzer eintragen
    // (bewusst ohne PIN – in der Gemeinde vertrauen wir einander)
    // Ausnahme: das Profil des Verwalters (einstellungen/entwickler) – sonst wäre jedes Gerät Verwalter
    function perAuswahl(pid) {
      return 'owners' in request.resource.data && ich() in request.resource.data.owners && !verwalterProfil(pid);
    }
    function verwalterProfil(pid) {
      let e = /databases/$(database)/documents/einstellungen/entwickler;
      return exists(e) && get(e).data.pid == pid;
    }
    // Zusammenführung, die der Verwalter in der Firebase-Konsole angelegt hat
    function perZusammenfuehrung(pid) {
      return exists(/databases/$(database)/documents/zusammenfuehren/$(ich()))
        && get(/databases/$(database)/documents/zusammenfuehren/$(ich())).data.zu == pid;
    }
    // Gültiger, nicht abgelaufener Kopplungscode für genau dieses Profil
    function perCode(pid) {
      return 'code' in request.resource.data
        && exists(/databases/$(database)/documents/kopplung/$(request.resource.data.code))
        && get(/databases/$(database)/documents/kopplung/$(request.resource.data.code)).data.pid == pid
        && get(/databases/$(database)/documents/kopplung/$(request.resource.data.code)).data.exp > request.time;
    }

    match /spieler/{pid} {
      allow read: if true;
      // Neuer Eintrag: Dokument-ID = Gerätekennung, oder – für weitere Spieler am selben Gerät –
      // Gerätekennung + "-" + 4 Zeichen (dann ist die Besitzerliste Pflicht)
      allow create: if angemeldet() && gueltig(request.resource.data) && (
          (pid == ich() && (!('owners' in request.resource.data) || request.resource.data.owners == [ich()]))
          || (pid.matches('^[A-Za-z0-9]+-[a-z0-9]{4}$') && pid.split('-')[0] == ich()
              && 'owners' in request.resource.data && request.resource.data.owners == [ich()]));
      // Werte nur EINMAL prüfen (Firestore erlaubt höchstens 1000 Rechenschritte pro Regel)
      // Beitreten (nur Besitzerliste/Kopplungscode ändern sich): Werte sind unverändert und werden nicht erneut
      // geprüft – bei vollen Profilen (alle Felder) überschritt die Prüfung sonst die 1000-Schritte-Grenze.
      allow update: if angemeldet() && besitzerBleiben() && (
          (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['owners', 'updated', 'code'])
            && (!('owners' in request.resource.data) || (request.resource.data.owners is list && request.resource.data.owners.size() <= 20))
            && text(request.resource.data, 'code', 12) && hoechstensIchDazu(pid)
            && (istBesitzer(pid) || perAuswahl(pid) || perZusammenfuehrung(pid) || perCode(pid)))
          || (werteOk(request.resource.data) && (
          (request.resource.data.keys().hasOnly(felder()) && hoechstensIchDazu(pid)
            && (istBesitzer(pid) || perAuswahl(pid) || perZusammenfuehrung(pid) || perCode(pid)))
          // Verwalter: doppelte Namen zusammenführen (Besitzerlisten bleiben unverändert)
          || (request.resource.data.keys().hasOnly(felder().concat(['zu'])) && zuOk(request.resource.data)
              && (!('owners' in resource.data) || request.resource.data.owners == resource.data.owners) && istEntwickler()))));
      allow delete: if angemeldet() && istBesitzer(pid);
    }

    // Kompletter Spielstand – nur für die Geräte des Spielers
    match /spielstand/{pid} {
      allow read, delete: if angemeldet() && besitzer(pid);
      allow create, update: if angemeldet() && besitzer(pid)
        && request.resource.data.keys().hasOnly(['daten', 'updated'])
        && request.resource.data.daten is string && request.resource.data.daten.size() <= 40000;
    }

    // Kopplungscodes: nur über den genauen Code lesbar, höchstens 30 Minuten gültig
    match /kopplung/{code} {
      allow get: if angemeldet();
      allow list: if false;
      allow create: if angemeldet() && code.size() == 6
        && request.resource.data.keys().hasOnly(['pid', 'exp'])
        && request.resource.data.pid is string && besitzer(request.resource.data.pid)
        && request.resource.data.exp is timestamp
        && request.resource.data.exp <= request.time + duration.value(30, 'm');
      allow delete: if angemeldet() && besitzer(resource.data.pid);
    }

    // Wer ist Verwalter? Das Dokument einstellungen/entwickler {pid: "<Spieler-ID>", geraete: ["<Gerätekennung>", …]}
    // pflegt der Verwalter NUR in der Firebase-Konsole (die App darf es nicht ändern). Verwalter-Rechte hat
    // ausschließlich ein Gerät, dessen Kennung in „geraete“ steht – nicht jedes Gerät des Spielerprofils.
    // Neues Gerät des Verwalters: Kennung unten im Profil ablesen und in der Konsole zu „geraete“ hinzufügen.
    function istEntwickler() {
      let e = /databases/$(database)/documents/einstellungen/entwickler;
      return angemeldet() && exists(e) && ich() in get(e).data.get('geraete', []);
    }
    match /einstellungen/{id} {
      allow get: if angemeldet();
    }

    // Feedback an den Entwickler: jedes angemeldete Gerät darf schreiben, nur der Entwickler lesen
    match /feedback/{id} {
      // Lesen, als erledigt markieren und löschen darf nur der Entwickler (Postfach in der App)
      allow read, delete: if istEntwickler();
      allow update: if istEntwickler() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['erledigt'])
        && request.resource.data.erledigt is bool;
      allow create: if angemeldet()
        && request.resource.data.keys().hasOnly(['text','art','ver','geraet','uid','name','pid','geschrieben','zeit'])
        && request.resource.data.text is string && request.resource.data.text.size() >= 3 && request.resource.data.text.size() <= 2000
        && request.resource.data.art in ['idee','fehler','lob','frage','info']
        && request.resource.data.uid == ich()
        && request.resource.data.zeit == request.time
        && text(request.resource.data, 'ver', 20) && text(request.resource.data, 'geraet', 80)
        && text(request.resource.data, 'name', 20) && text(request.resource.data, 'pid', 64) && text(request.resource.data, 'geschrieben', 40);
    }

    // ---------- Duelle (Version 6.31: allgemeines Modell; 6.32: Arten vers, kette, blitz) ----------
    // Ein Duell-Dokument: { v: Schema-Version, art: "vers" | später z. B. "gruppe", modus: "1v1" | später "team",
    //   von: Ersteller-pid, teil: [pid, …] (Ersteller zuerst; Abfrage per array-contains), namen: {pid: Name},
    //   seiten: {pid: "a"|"b"} (nur Team-Varianten), opt: {kleine Zusatzwerte der Variante},
    //   k: Lernvers-Monat (vers) bzw. Kette bibel|at|nt|rueck (kette) bzw. Bereich all (blitz), seed: Startwert,
    //   ergebnisse: {pid: {p, r, n}}, am: pid des letzten Schreibers, status: offen|fertig|abgelehnt, erstellt, ablauf, fertig }
    // Jede Variante wird unten in artErlaubt() einzeln freigeschaltet – heute Lernvers-, Ketten-, Blitz- und Biblionär-Duell (quiz/leiter, bis 1.000.000 Punkte) eins gegen eins.
    function ergOk(e) {
      return e is map && e.keys().hasOnly(['p', 'r', 'n']) && e.p is int && e.p >= 0 && e.p <= 1000000
        && e.r is int && e.r >= 0 && e.r <= 500 && e.n is int && e.n >= 1 && e.n <= 500 && e.r <= e.n;
    }
    function artErlaubt(d) {
      return (d.art == 'vers' || (d.art == 'kette' && d.k in ['bibel', 'at', 'nt', 'rueck']) || (d.art == 'blitz' && d.k == 'all') || (d.art == 'quiz' && d.k == 'leiter'))
        && d.modus == '1v1' && d.teil.size() == 2 && !('seiten' in d)
        && exists(spielerPfad(d.teil[1]));
    }
    function duellNeu(d) {
      return d.keys().hasOnly(['v', 'art', 'modus', 'von', 'teil', 'namen', 'seiten', 'opt', 'k', 'seed', 'ergebnisse', 'status', 'erstellt', 'ablauf'])
        && d.v is int && d.v >= 1 && d.v <= 99 && d.art is string && d.art.size() <= 20 && d.modus is string && d.modus.size() <= 20
        && d.von is string && d.teil is list && d.teil.size() >= 2 && d.teil.size() <= 20 && d.teil[0] == d.von && d.teil[1] != d.von
        && d.namen is map && d.namen.size() <= 20
        && (!('seiten' in d) || (d.seiten is map && d.seiten.size() <= 20))
        && (!('opt' in d) || (d.opt is map && d.opt.size() <= 10))
        && d.k is string && d.k.size() <= 20 && d.seed is int && d.seed >= 0 && d.seed < 2147483648
        && d.ergebnisse is map && d.ergebnisse.keys().hasOnly([d.von]) && ergOk(d.ergebnisse[d.von])
        && d.status == 'offen' && d.erstellt == request.time
        && d.ablauf is timestamp && d.ablauf <= request.time + duration.value(8, 'd')
        && artErlaubt(d) && besitzer(d.von);
    }
    function neu() { return request.resource.data; }
    match /duelle/{id} {
      allow read: if angemeldet();
      allow create: if angemeldet() && duellNeu(request.resource.data);
      allow update: if angemeldet() && resource.data.status == 'offen' && neu().am is string && neu().am in resource.data.teil && (
          // Ein Teilnehmer trägt EINMAL sein eigenes Ergebnis ein; sind alle fertig, wird das Duell abgeschlossen
          (neu().diff(resource.data).affectedKeys().hasOnly(['ergebnisse', 'am', 'status', 'fertig'])
            && !(neu().am in resource.data.ergebnisse)
            && neu().ergebnisse.diff(resource.data.ergebnisse).affectedKeys().hasOnly([neu().am])
            && ergOk(neu().ergebnisse[neu().am])
            && ((neu().ergebnisse.keys().hasAll(resource.data.teil) && neu().status == 'fertig' && neu().fertig == request.time)
              || (!neu().ergebnisse.keys().hasAll(resource.data.teil) && neu().status == 'offen' && !('fertig' in neu()))))
          // Ablehnen kann nur, wer noch nicht gespielt hat
          || (neu().diff(resource.data).affectedKeys().hasOnly(['am', 'status']) && neu().status == 'abgelehnt'
            && !(neu().am in resource.data.ergebnisse))
        ) && besitzer(neu().am);
      // Löschen: Ersteller oder (beim Duell eins gegen eins) der Gegner – aber nie fertige Duelle (daraus wird die Duell-Rangliste berechnet)
      allow delete: if angemeldet() && resource.data.status != 'fertig' && (besitzer(resource.data.von) || besitzer(resource.data.teil[1]));
    }

    // „Biblionär der Woche“ (6.40): ein Dokument je Woche und Spieler, ID = Woche + "_" + pid.
    // Bewusst eigene Sammlung – die Regel für spieler liegt nahe an der Grenze von 1000 Rechenschritten.
    function bibOk(id) {
      let d = request.resource.data;
      return d.keys().hasOnly(['pid', 'wk', 'p', 'name', 'av', 'fr', 'lv', 'updated'])
        && d.pid is string && d.wk is string && d.wk.size() <= 10 && id == d.wk + '_' + d.pid
        && d.p is int && d.p >= 0 && d.p <= 1000000
        && d.name is string && d.name.size() >= 2 && d.name.size() <= 20
        && text(d, 'av', 20) && text(d, 'fr', 20) && zahl(d, 'lv', 1000)
        && besitzer(d.pid);
    }
    match /biblionaer/{id} {
      allow read: if true;
      allow create: if angemeldet() && bibOk(id);
      allow update: if angemeldet() && bibOk(id) && request.resource.data.pid == resource.data.pid;
      allow delete: if angemeldet() && besitzer(resource.data.pid);
    }

    // 6.68: Gemeinsamer Bau der Stiftshütte – je Spieler ein Dokument mit seinen Spenden je Bauabschnitt (1–8).
    // Spenden können nur steigen (nichts wird zurückgenommen), höchstens 20.000 je Abschnitt.
    function spende(k) {
      let n = request.resource.data.s;
      return !(k in n) || (n[k] is int && n[k] >= 0 && n[k] <= 20000
        && (resource == null || !(k in resource.data.s) || n[k] >= resource.data.s[k]));
    }
    // 7.3: Weg nach Bethlehem (Advent) – wie die Stiftshütte; s = Talente-Spenden, g = tägliche Wegzehrung. Nur steigend, höchstens 20.000.
    function gabeFeld(feld, k) {
      let n = request.resource.data[feld];
      return !(k in n) || (n[k] is int && n[k] >= 0 && n[k] <= 20000
        && (resource == null || !(feld in resource.data) || !(k in resource.data[feld]) || n[k] >= resource.data[feld][k]));
    }
    function gabeOk(feld) {
      return request.resource.data[feld] is map && request.resource.data[feld].keys().hasOnly(['1', '2', '3', '4', '5', '6', '7', '8'])
        && (resource == null || !(feld in resource.data) || request.resource.data[feld].keys().hasAll(resource.data[feld].keys()))
        && gabeFeld(feld, '1') && gabeFeld(feld, '2') && gabeFeld(feld, '3') && gabeFeld(feld, '4') && gabeFeld(feld, '5') && gabeFeld(feld, '6') && gabeFeld(feld, '7') && gabeFeld(feld, '8');
    }
    match /bethlehem/{pid} {
      allow read: if true;
      allow create, update: if angemeldet() && besitzer(pid)
        && request.resource.data.keys().hasOnly(['pid', 'name', 's', 'g', 'updated'])
        && request.resource.data.pid == pid
        && request.resource.data.name is string && request.resource.data.name.size() >= 2 && request.resource.data.name.size() <= 20
        && gabeOk('s') && gabeOk('g');
      allow delete: if false;
    }
    match /stiftshuette/{pid} {
      allow read: if true;
      allow create, update: if angemeldet() && besitzer(pid)
        && request.resource.data.keys().hasOnly(['pid', 'name', 's', 'updated'])
        && request.resource.data.pid == pid
        && request.resource.data.name is string && request.resource.data.name.size() >= 2 && request.resource.data.name.size() <= 20
        && request.resource.data.s is map && request.resource.data.s.keys().hasOnly(['1', '2', '3', '4', '5', '6', '7', '8'])
        && (resource == null || request.resource.data.s.keys().hasAll(resource.data.s.keys()))
        && spende('1') && spende('2') && spende('3') && spende('4') && spende('5') && spende('6') && spende('7') && spende('8');
      allow delete: if false;
    }

    // 6.73: XP-Quellen – je Spieler ein Dokument, das zeigt, woher die XP kommen (Quelle → XP/Anzahl, größte Einzelvergabe,
    // XP je Tag und je Stunde). Summen können nur steigen, der Anfangsbestand „alt“ nur sinken – so lässt sich die Zählung
    // nicht nachträglich „glätten“. Lesen darf jeder angemeldete Spieler wie die Rangliste; ausgewertet wird sie in der XP-Prüfung des Verwalters.
    function xqKeys() { return ['vers','lektion','luecke','blitz','kette','duell','duelleigen','bib','tagesbonus','spickzettel','sonst','abgleich','spende']; }
    // 6.88: Die Regel war so aufwendig, dass Firestore bei vier oder mehr Quellen mit „maximum of 1000 expressions“ ablehnte (alle Schreibzugriffe
    // der XP-Zählung nach den ersten drei Quellen, bei allen Spielern). Jetzt: Ausdrücke einmal in „let“ gebunden, die Schlüssel von q, n und m
    // in einem Aufruf geprüft, Werte von n/m nicht mehr einzeln (sie sind nur Zähler für die Anzeige). Geprüft bleiben: Summen nur steigend
    // und ganzzahlig, Anfangsbestand nur sinkend, Belohnungs-Protokoll nie kürzer, nur bekannte Quellen.
    function xqSumme(k, q, rq) {
      return !(k in q) || (q[k] is int && q[k] >= 0 && q[k] <= 10000000 && (!(k in rq) || q[k] >= rq[k]));
    }
    function xqGueltig(pid) {
      let d = request.resource.data;
      let q = d.q;
      let rq = resource == null ? {} : resource.data.q;
      return d.keys().hasOnly(['pid', 'name', 'alt', 'q', 'n', 'm', 'tag', 'std', 'bel', 'updated'])
        && d.pid == pid
        && d.name is string && d.name.size() >= 2 && d.name.size() <= 20
        && d.alt is int && d.alt >= 0 && d.alt <= 10000000
        && (resource == null || d.alt <= resource.data.alt)
        && q is map && d.n is map && d.m is map
        && q.keys().concat(d.n.keys()).concat(d.m.keys()).hasOnly(xqKeys())
        && d.tag is map && d.tag.size() <= 12
        && d.std is map && d.std.size() <= 50
        // Protokoll der täglichen Belohnung: höchstens 100 Einträge, es darf nie kürzer werden, bis die 100 erreicht sind
        && (!('bel' in d) || (d.bel is list && d.bel.size() <= 100))
        && (resource == null || !('bel' in resource.data) || ('bel' in d && d.bel.size() >= resource.data.bel.size()))
        && xqSumme('vers', q, rq) && xqSumme('lektion', q, rq) && xqSumme('luecke', q, rq) && xqSumme('blitz', q, rq) && xqSumme('kette', q, rq)
        && xqSumme('duell', q, rq) && xqSumme('duelleigen', q, rq) && xqSumme('bib', q, rq) && xqSumme('tagesbonus', q, rq)
        && xqSumme('spickzettel', q, rq) && xqSumme('sonst', q, rq) && xqSumme('abgleich', q, rq) && xqSumme('spende', q, rq);
    }
    // 6.88: „Zählung anfordern“ – nur der Verwalter legt hier eine Markierung {ts} an (Dokument <Spieler-ID> oder _alle).
    // Die App des Spielers gleicht dann beim nächsten Start seine XP-Zählung ab. Lesen darf jedes angemeldete Gerät.
    match /abgleich/{id} {
      allow read: if angemeldet();
      allow create, update: if istEntwickler() && request.resource.data.keys().hasOnly(['ts']) && request.resource.data.ts is int;
      allow delete: if false;
    }
    // 6.95: Talente-Zählung für die Verwalter-Seite „Talente-Planung“. Schreiben darf nur der Besitzer des Profils, lesen nur der Verwalter.
    //       t = verdiente Talente je Tag (Schlüssel = Tagesnummer, höchstens 45 Tage), sd = Spieltage (höchstens 40).
    function tlGueltig(pid) {
      let d = request.resource.data;
      return d.keys().hasOnly(['pid', 'name', 'tal', 's', 'v', 'a', 'p', 't', 'sd', 'updated'])
        && d.pid == pid
        && d.name is string && d.name.size() >= 2 && d.name.size() <= 20
        && d.tal is int && d.tal >= 0 && d.tal <= 100000000
        && d.s is int && d.s >= 0 && d.s <= 1000000
        && d.v is int && d.v >= 0 && d.v <= 100000000
        && d.a is int && d.a >= 0 && d.a <= 100000000
        && d.p is int && d.p >= 0 && d.p <= 100000000
        && d.t is map && d.t.size() <= 45
        && d.sd is list && d.sd.size() <= 40;
    }
    match /talente/{pid} {
      allow read: if istEntwickler();
      allow create, update: if angemeldet() && besitzer(pid) && tlGueltig(pid);
      allow delete: if false;
    }
    match /xpquellen/{pid} {
      allow read: if angemeldet();
      allow create, update: if angemeldet() && besitzer(pid) && xqGueltig(pid);
      allow delete: if false;
    }

    // 6.75: XP-Protokoll – jede XP-Vergabe ein Dokument <pid>_<Zeit>_<Nr>. Nur anlegen: Ändern und Löschen ist niemandem erlaubt, auch dem
    // Verwalter nicht – so bleibt das Protokoll nachprüfbar. Die Werte dürfen bewusst groß sein (auch Auffälliges soll sichtbar bleiben);
    // „gebucht“ ist die Uhrzeit des Servers, „z“ die der Geräteuhr. Lesen darf nur der Verwalter.
    match /xpprotokoll/{id} {
      allow read: if istEntwickler();
      allow create: if angemeldet()
        && request.resource.data.keys().hasOnly(['pid', 'name', 'z', 'q', 'x', 'v', 's', 'f', 'fm', 'd', 'ver', 'gebucht'])
        && request.resource.data.keys().hasAll(['pid', 'z', 'q', 'x', 'v', 'gebucht'])
        && request.resource.data.pid is string && request.resource.data.pid.size() >= 4 && request.resource.data.pid.size() <= 64
        && id.matches(request.resource.data.pid + '_[0-9]+_[0-9]+')
        && besitzer(request.resource.data.pid)
        && request.resource.data.q in xqKeys()
        && request.resource.data.z is int && request.resource.data.z <= request.time.toMillis() + 600000
        && request.resource.data.x is int && request.resource.data.x > 0 && request.resource.data.x <= 10000000
        && request.resource.data.v is int && request.resource.data.v >= 0 && request.resource.data.v <= 100000000
        && request.resource.data.gebucht == request.time
        && (!('name' in request.resource.data) || (request.resource.data.name is string && request.resource.data.name.size() <= 20))
        && (!('s' in request.resource.data) || (request.resource.data.s is int && request.resource.data.s >= 0 && request.resource.data.s <= 86400))
        && (!('f' in request.resource.data) || (request.resource.data.f is int && request.resource.data.f >= 0 && request.resource.data.f <= 100000))
        && (!('fm' in request.resource.data) || (request.resource.data.fm is int && request.resource.data.fm >= 0 && request.resource.data.fm <= 100000))
        && (!('d' in request.resource.data) || (request.resource.data.d is string && request.resource.data.d.size() <= 60))
        && (!('ver' in request.resource.data) || (request.resource.data.ver is string && request.resource.data.ver.size() <= 12));
      allow update, delete: if false;
    }

    // Zusammenführungen legt nur der Verwalter in der Konsole an; jedes Gerät darf nur seine eigene lesen
    // 6.78: Schatztruhe – je Spieler und Serverdatum höchstens ein Dokument <pid>_<JJJJMMTT> (nur anlegen, nie ändern/löschen).
    // Das Datum muss zur Serverzeit passen (heute oder gestern, Deutschland = UTC+1/+2) – Uhr am Gerät umstellen hilft nicht.
    // Lesen darf jeder Angemeldete einzeln (Prüfung „schon geholt?“), Auflisten nur der Verwalter.
    function truheTag(t) { return t.year() * 10000 + t.month() * 100 + t.day(); }
    function truheTagOk(tag) {
      let a = request.time + duration.value(1, 'h');
      let b = request.time + duration.value(2, 'h');
      return tag in [truheTag(a), truheTag(b), truheTag(a - duration.value(1, 'd')), truheTag(b - duration.value(1, 'd'))];
    }
    match /tagestruhe/{id} {
      allow get: if angemeldet();
      allow list: if istEntwickler();
      allow create: if angemeldet()
        && request.resource.data.keys().hasOnly(['pid', 'name', 'tag', 'z', 'ver', 'gebucht'])
        && request.resource.data.keys().hasAll(['pid', 'tag', 'gebucht'])
        && request.resource.data.pid is string && request.resource.data.pid.size() >= 4 && request.resource.data.pid.size() <= 64
        && request.resource.data.tag is int
        && id == request.resource.data.pid + '_' + string(request.resource.data.tag)
        && truheTagOk(request.resource.data.tag)
        && besitzer(request.resource.data.pid)
        && request.resource.data.gebucht == request.time
        && (!('name' in request.resource.data) || (request.resource.data.name is string && request.resource.data.name.size() <= 20))
        && (!('z' in request.resource.data) || request.resource.data.z is int)
        && (!('ver' in request.resource.data) || (request.resource.data.ver is string && request.resource.data.ver.size() <= 12));
      allow update, delete: if false;
    }

    // 6.80: Wochen-Sicherung „Biblionär der Woche“ – je Spieler und Serverwoche höchstens ein Dokument <pid>_<Wochennummer> (nur anlegen).
    // Wochennummer = Wochen seit Montag, 5.1.2026 (Deutschland = UTC+1/+2, daher beide Werte erlaubt); die Serverzeit entscheidet.
    function wochenNr(h) { return math.floor((request.time.toMillis() + h * 3600000 - 1767571200000) / 604800000); }
    match /bibversuch/{id} {
      allow get: if angemeldet();
      allow list: if istEntwickler();
      allow create: if angemeldet()
        && request.resource.data.keys().hasOnly(['pid', 'name', 'w', 'wk', 'z', 'ver', 'gebucht'])
        && request.resource.data.keys().hasAll(['pid', 'w', 'gebucht'])
        && request.resource.data.pid is string && request.resource.data.pid.size() >= 4 && request.resource.data.pid.size() <= 64
        && request.resource.data.w is int
        && id == request.resource.data.pid + '_' + string(request.resource.data.w)
        && (request.resource.data.w == wochenNr(1) || request.resource.data.w == wochenNr(2))
        && besitzer(request.resource.data.pid)
        && request.resource.data.gebucht == request.time
        && (!('name' in request.resource.data) || (request.resource.data.name is string && request.resource.data.name.size() <= 20))
        && (!('wk' in request.resource.data) || (request.resource.data.wk is string && request.resource.data.wk.size() <= 10))
        && (!('z' in request.resource.data) || request.resource.data.z is int)
        && (!('ver' in request.resource.data) || (request.resource.data.ver is string && request.resource.data.ver.size() <= 12));
      allow update, delete: if false;
    }

    // 6.80: XP für Spenden an Events (Stiftshütte) – je Spieler und Abschnitt ein Dokument <pid>_stift<1-8> mit den Talenten, für die schon
    // XP gezahlt wurden. Nur steigend, nie mehr als in der Stiftshütte wirklich gespendet ist – jedes Talent bringt so nur einmal XP.
    function stiftGabe(pid, k) { return get(/databases/$(database)/documents/stiftshuette/$(pid)).data.s.get(k, 0); }
    function spendenXpOk(id) {
      let d = request.resource.data;
      return d.keys().hasOnly(['pid', 'name', 'k', 'tal', 'z', 'ver', 'updated'])
        && d.keys().hasAll(['pid', 'k', 'tal', 'updated'])
        && d.pid is string && d.pid.size() >= 4 && d.pid.size() <= 64
        && d.k in ['1', '2', '3', '4', '5', '6', '7', '8']
        && id == d.pid + '_stift' + d.k
        && d.tal is int && d.tal > 0 && d.tal <= 20000 && d.tal <= stiftGabe(d.pid, d.k)
        && besitzer(d.pid) && d.updated == request.time
        && (!('name' in d) || (d.name is string && d.name.size() <= 20))
        && (!('z' in d) || d.z is int)
        && (!('ver' in d) || (d.ver is string && d.ver.size() <= 12));
    }
    match /spendenxp/{id} {
      allow get: if angemeldet();
      allow list: if istEntwickler();
      allow create: if angemeldet() && spendenXpOk(id);
      allow update: if angemeldet() && spendenXpOk(id) && request.resource.data.tal >= resource.data.tal && request.resource.data.pid == resource.data.pid;
      allow delete: if false;
    }

    match /zusammenfuehren/{uid} {
      allow get: if angemeldet() && (ich() == uid || istEntwickler());
      // Der Verwalter (Entwickler) legt Zusammenführungen automatisch aus der App an
      allow create, update: if istEntwickler() && request.resource.data.keys().hasOnly(['zu'])
        && request.resource.data.zu is string && request.resource.data.zu.size() <= 64;
    }
  }
