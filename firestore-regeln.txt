rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /spieler/{uid} {
      // Jeder darf die Rangliste lesen
      allow read: if true;
      // Jeder darf nur seinen eigenen Eintrag löschen
      allow delete: if request.auth != null && request.auth.uid == uid;
      // Jeder darf nur seinen eigenen Eintrag schreiben, mit plausiblen Werten
      allow create, update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['name','xp','level','bestChain','stars','badges','streak','week','weekXp','updated'])
        && request.resource.data.name is string
        && request.resource.data.name.size() >= 2 && request.resource.data.name.size() <= 20
        && request.resource.data.xp is int && request.resource.data.xp >= 0 && request.resource.data.xp <= 1000000
        && request.resource.data.weekXp is int && request.resource.data.weekXp >= 0 && request.resource.data.weekXp <= 100000
        && request.resource.data.bestChain is int && request.resource.data.bestChain >= 0 && request.resource.data.bestChain <= 65
        && request.resource.data.level is int && request.resource.data.level >= 1 && request.resource.data.level <= 500
        && request.resource.data.stars is int && request.resource.data.stars >= 0 && request.resource.data.stars <= 24
        && request.resource.data.badges is int && request.resource.data.badges >= 0 && request.resource.data.badges <= 50
        && request.resource.data.streak is int && request.resource.data.streak >= 0 && request.resource.data.streak <= 10000
        && request.resource.data.week is string && request.resource.data.week.size() <= 10;
    }
  }
}
